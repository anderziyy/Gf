local BASE_VIEWPORT_SIZE = Vector2.new(1600, 720)

local function getResponsiveScale()
    local camera = workspace.CurrentCamera
    local viewportSize = camera and camera.ViewportSize or BASE_VIEWPORT_SIZE
    local scale = math.min(viewportSize.X / BASE_VIEWPORT_SIZE.X, viewportSize.Y / BASE_VIEWPORT_SIZE.Y)
    return math.clamp(scale, 0.85, 2.15)
end

local function rpx(value)
    if value == 0 then
        return 0
    end

    local scale = getResponsiveScale()
    local sign = value < 0 and -1 or 1
    return sign * math.max(1, math.floor(math.abs(value) * scale + 0.5))
end

local fileName = "Phantom_" .. tostring(Player.UserId) .. ".json"
local savedSettings = {}

if type(isfile) == "function" and isfile(fileName) then
    local success, data = pcall(function()
        return HttpService:JSONDecode(readfile(fileName))
    end)
    if success and type(data) == "table" then
        savedSettings = data
    end
end

local function saveSettings()
    if type(writefile) == "function" then
        pcall(function()
            writefile(fileName, HttpService:JSONEncode(savedSettings))
        end)
    end
end

local optionStates = {}

local function createOptionState(name, initialValue)
    if savedSettings[name] ~= nil then
        optionStates[name] = savedSettings[name]
    else
        optionStates[name] = initialValue
    end
    return function() return optionStates[name] end
end

local function setOptionState(name, value)
    optionStates[name] = value
    savedSettings[name] = value
    saveSettings()
end

local function saveCurrentSettings()
    for name, value in pairs(optionStates) do
        if name ~= "_debounce" then
            savedSettings[name] = value
        end
    end

    saveSettings()
end

local function getContentPadding()
    local height = MainContent.AbsoluteSize.Y
    if height > 0 then
        return math.max(1, math.floor(height * 0.022 + 0.5))
    end
    return rpx(8)
end

local function getDropdownPadding()
    return math.max(1, math.floor(getContentPadding() * 0.5 + 0.5))
end

local function getDropdownContentHeight(count)
    return (count * getOptionFrameHeight()) + (math.max(count - 1, 0) * getDropdownPadding())
end

local function getDropdownVisibleHeight(count, maxVisible)
    return getDropdownContentHeight(math.min(count, maxVisible))
end

local function getCategoryKey(tabName, categoryName)
    return tabName .. "::" .. categoryName
end

local function registerElement(tabName, element, elementName)
    if not tabs[tabName] then
        tabs[tabName] = { button = nil, elements = {} }
    end

    table.insert(tabs[tabName].elements, element)
    optionElementsMap[element] = elementName or ""
    optionTabMap[element] = tabName

    local categoryName = currentCategoryByTab[tabName]
    if categoryName then
        local key = getCategoryKey(tabName, categoryName)
        elementCategoryMap[element] = key
        if not categoryChildren[key] then
            categoryChildren[key] = {}
        end
        table.insert(categoryChildren[key], element)
    end
end

			progressLine.Size = UDim2.new(totalWidth * progress, rpx(0), 0.045, rpx(0))
		end

		if progress >= 1 then
			progressConnection:Disconnect()

			if notificationGui and notificationGui.Parent then
				local slideOutTween = TweenService:Create(
					notificationGui,
					TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
					{Position = UDim2.new(0.99, rpx(0), 0.8, rpx(0))}
				)
				slideOutTween:Play()

				slideOutTween.Completed:Connect(function()
					if notificationGui then
						notificationGui:Destroy()
					end
					if notificationScreenGui then
						notificationScreenGui:Destroy()
					end
					notificationTimers[notificationId] = nil
				end)
			end
		end
	end)
end

local function createDebounce(key, duration)
    if not optionStates["_debounce"] then optionStates["_debounce"] = {} end
    if optionStates["_debounce"][key] then return false end
    optionStates["_debounce"][key] = true
    task.delay(duration or 0.05, function() optionStates["_debounce"][key] = nil end)
    return true
end

local function isPointInsideGui(guiObject, position)
    if not guiObject or not guiObject.Parent or not guiObject.Visible then
        return false
    end

    local absolutePosition = guiObject.AbsolutePosition
    local absoluteSize = guiObject.AbsoluteSize
    return position.X >= absolutePosition.X and position.X <= absolutePosition.X + absoluteSize.X and position.Y >= absolutePosition.Y and position.Y <= absolutePosition.Y + absoluteSize.Y
end

			local function VerificarCapacidadeMochila()
				local backpackCapacity = "0/5"
				if LocalPlayer:FindFirstChild("PlayerGui") then
					local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
					local mainScreenGui = playerGui:FindFirstChild("MainScreenGui")
					if mainScreenGui then
						local currencies = mainScreenGui:FindFirstChild("Currencies")
						if currencies then
							local backpack = currencies:FindFirstChild("Backpack")
							if backpack and backpack:FindFirstChild("Value") then
								backpackCapacity = backpack.Value.Text
							end
						end
					end
				end
				
				local currentCap, maxCap = backpackCapacity:match("(%d+)/(%d+)")
				currentCap = tonumber(currentCap) or 0
				maxCap = tonumber(maxCap) or 5
				return currentCap, maxCap, backpackCapacity

			local function ObterRaridade(item)
				if not item then
					return nil
				end

				-- Primeiro tenta o LootGui diretamente no item.
				local lootGui = item:FindFirstChild("LootGui")
				if lootGui and lootGui:IsA("BillboardGui") then
					local rarity = lootGui:FindFirstChild("Rarity")
					if rarity and rarity:IsA("TextLabel") and rarity.Text ~= "" then
						return string.lower(rarity.Text)
					end
				end

				-- Compatibilidade com a estrutura em que o LootGui fica na Base.
			if item:IsA("Model") then
					local base = item:FindFirstChild("Base")
					if base then
						lootGui = base:FindFirstChild("LootGui")
						if lootGui and lootGui:IsA("BillboardGui") then
							local rarity = lootGui:FindFirstChild("Rarity")
							if rarity and rarity:IsA("TextLabel") and rarity.Text ~= "" then
								return string.lower(rarity.Text)
							end
						end
					end
				end

				-- Último fallback: tenta identificar a raridade pelo nome do loot.
				for _, atributo in pairs(LootConfig.Rarities) do
					if string.find(string.lower(item.Name or ""), string.lower(atributo), 1, true) then
						return string.lower(atributo)
					end
				end

				return nil
			end
			
			local function VerificarSeRaridadeEValida(raridade, mundo)
				if not raridade or not mundo then return false end

			local function LoopUnificado()
				while task.wait(0.1) do
					if not autoPlayersActive then break end
					
					AtualizarInfoMundo()
					
					Character = LocalPlayer.Character
					if not Character then
						task.wait(0.5)
						continue
					end
					HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
					if not HumanoidRootPart then continue end
					
					local mundoIdeal = ObterMundoIdeal()
					if CurrentWorld ~= mundoIdeal then
						TeleportarParaMundo(mundoIdeal)
						task.wait(2)
						continue
					end
					
					local currentCap, maxCap, backpackCapacity = VerificarCapacidadeMochila()
					
					if currentCap >= maxCap then
						pcall(function()
							local Event = game:GetService("ReplicatedStorage").Packages._Index["acecateer_knit@1.7.2"].knit.Services.BaseTeleportService.RF.TeleportToSpawn
							if Event then
								Event:InvokeServer()
							end
						end)
						
						while autoPlayersActive do
							task.wait(0.5)
							local capAtual, _ = VerificarCapacidadeMochila()
							if capAtual < maxCap then
								precisaTeleportarZona = true
								break
							end
						end
						continue
					end
					
					if precisaTeleportarZona then
						local coordDestino = COORDENADAS_ZONA_INICIAL[CurrentWorld] or COORDENADAS_ZONA_INICIAL[1]
						HumanoidRootPart.CFrame = coordDestino
						precisaTeleportarZona = false
						task.wait(0.5)
					end
					
					if not VerificarSeItemExiste(ItemAtualSelecionado) then
						ItemAtualSelecionado = nil
					end
					
					local todasAsZonas = ObterTodasAsZonas()
					local itemTeleportado = false
					
					if ItemAtualSelecionado and VerificarSeItemExiste(ItemAtualSelecionado) then
						local pecaBase = ObterPecaBase(ItemAtualSelecionado)
						if pecaBase then
							HumanoidRootPart.CFrame = CFrame.new(pecaBase.Position + Vector3.new(0, 3, 0))
							itemTeleportado = true
						end
					elseif #todasAsZonas > 0 then
						local itensPorRaridade = {}
						local allItems = {}
						
						for _, zona in pairs(todasAsZonas) do
							for _, descendente in pairs(zona:GetDescendants()) do
								if descendente:IsA("Model") or descendente:IsA("BasePart") then
									local nome = descendente.Name
									if string.match(nome, "^%d+_") and descendente:IsA("Model") then
										local pecaBase = ObterPecaBase(descendente)
										if pecaBase then
											local distancia = (HumanoidRootPart.Position - pecaBase.Position).Magnitude
											if distancia > 5 then
												local raridade = ObterRaridade(descendente)
												
												if VerificarSeRaridadeEValida(raridade, CurrentWorld) then
													local valorRaridade = ObterValorRaridade(raridade, CurrentWorld)
													
													if not itensPorRaridade[valorRaridade] then
														itensPorRaridade[valorRaridade] = {}
													end
													
													table.insert(itensPorRaridade[valorRaridade], {
														item = descendente,
														parte = pecaBase,
														distancia = distancia,
														raridade = raridade,
														valor = valorRaridade
													})
													table.insert(allItems, descendente)
												end
											end
										end
									end
								end
							end
						end
						
						local itemMaisProximo = nil
						local maxHierarquia = 0
						
						for valor, _ in pairs(itensPorRaridade) do
							if valor > maxHierarquia then
								maxHierarquia = valor
							end
						end
						
						if itensPorRaridade[maxHierarquia] and #itensPorRaridade[maxHierarquia] > 0 then
							local itensComMenorDistancia = itensPorRaridade[maxHierarquia]
							table.sort(itensComMenorDistancia, function(a, b)
								return a.distancia < b.distancia
							end)
							itemMaisProximo = itensComMenorDistancia[1].item
						end
						
						if itemMaisProximo then
							ItemAtualSelecionado = itemMaisProximo
							local pecaBase = ObterPecaBase(itemMaisProximo)
							if pecaBase then
								HumanoidRootPart.CFrame = CFrame.new(pecaBase.Position + Vector3.new(0, 3, 0))
								itemTeleportado = true
							end
						end
					end
					
					local prompts = BuscarTodosOsPrompts()
					local coletado = false
					
					for _, prompt in pairs(prompts) do
						if autoPlayersActive and prompt:IsA("ProximityPrompt") and prompt.Enabled then
							local nome = prompt.Name or ""
							local acao = prompt.ActionText or ""
							
							local ePickup = 
								nome == "PickupPrompt" or
								string.find(nome, "Pickup") or
								string.find(nome, "Pick") or
								string.find(nome, "Collect") or
								string.find(nome, "Coletar") or
								string.find(nome, "Pegar") or
								string.find(nome, "Grab") or
								string.find(acao, "Pickup") or
								string.find(acao, "Pick up") or
								string.find(acao, "Coletar") or
								string.find(acao, "Pegar") or
								string.find(acao, "Grab") or
								string.find(acao, "Collect") or
								prompt:GetAttribute("LootPickupPrompt") == true
							
							if ePickup then
								local part = prompt.Parent
								if part then
									local posicao = nil
									if part:IsA("BasePart") then
										posicao = part.Position
									elseif part:IsA("Attachment") then
										posicao = part.WorldPosition
									end
									
									if posicao then
										local distancia = (HumanoidRootPart.Position - posicao).Magnitude
										if distancia <= prompt.MaxActivationDistance then
											coletado = true
											pcall(function() prompt.HoldDuration = 0 end)
											pcall(function() fireproximityprompt(prompt) end)
											pcall(function() prompt:Trigger(LocalPlayer) end)
											pcall(function() ProximityPromptService:PromptTriggered(prompt, LocalPlayer) end)
											pcall(function() prompt:InputHoldBegin() task.wait(0.01) prompt:InputHoldEnd() end)
										end
									end
								end
							end
						end
					end
				end
			end
			
			LoopUnificado()
		end)
	end
end)

local function getGrassLineSelectedWorldNumber()
    local selected = optionStates["GrassLineWorldSelector"]
    if selected == nil then
        return nil
    end

    local number = tonumber(string.match(tostring(selected), "%d+"))
    if not number then
        return nil
    end

    return math.clamp(number, 1, 4)
end

local function getGrassLineCurrentWorld()
    return tonumber(Player:GetAttribute("CurrentWorld")) or 1
end

local function getGrassLineUnlockedWorlds()
    local highest = tonumber(Player:GetAttribute("HighestUnlockedWorld")) or 1
    return math.max(1, math.floor(highest))
end

local function teleportGrassLineToWorld(worldNumber)
    local success = false

    pcall(function()
        local Event = ReplicatedStorage.Packages._Index["acecateer_knit@1.7.2"].knit.Services.WorldService.RF.TeleportToWorld
        if Event then
            Event:InvokeServer(worldNumber)
            success = true
        end
    end)

    if success then
        local timeout = 0
        repeat
            task.wait(0.1)
            timeout += 0.1
        until not autoFarmGrassLineActive or getGrassLineCurrentWorld() == worldNumber or timeout >= 4
    end

    return success
end

local function executeGrassLineSpawnScript()
    pcall(function()
        local Event = ReplicatedStorage.Packages._Index["acecateer_knit@1.7.2"].knit.Services.BaseTeleportService.RF.TeleportToSpawn
        if Event then
            Event:InvokeServer()
        end
    end)
end

local function setGrassLineSpeedForHumanoid(humanoid)
    if not humanoid or not humanoid.Parent then
        return
    end

    if grassLineOriginalSpeeds[humanoid] == nil then
        grassLineOriginalSpeeds[humanoid] = humanoid.WalkSpeed
    end

    humanoid.WalkSpeed = 60
end

local function restoreGrassLineHumanoidSpeed(humanoid)
    if not humanoid then
        return
    end

    local original = grassLineOriginalSpeeds[humanoid]
    if original ~= nil then
        pcall(function()
            humanoid.WalkSpeed = original
        end)
        grassLineOriginalSpeeds[humanoid] = nil
    elseif humanoid.Parent then
        pcall(function()
            humanoid.WalkSpeed = 28
        end)
    end
end

local function disconnectGrassLineSpeedConnections()
    for _, connection in pairs(grassLineSpeedConnections) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    table.clear(grassLineSpeedConnections)
end

local function setupGrassLineCharacterSpeed()
    disconnectGrassLineSpeedConnections()

    local character = Player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        setGrassLineSpeedForHumanoid(humanoid)
    end

    grassLineSpeedConnections.characterAdded = Player.CharacterAdded:Connect(function(characterAdded)
        if not autoFarmGrassLineActive then
            return
        end

        local humanoidAdded = characterAdded:WaitForChild("Humanoid", 5)
        if humanoidAdded then
            setGrassLineSpeedForHumanoid(humanoidAdded)
        end
    end)
end

local function walkGrassLineToCoordinate(targetCFrame, targetWorldNumber)
    local targetPosition = targetCFrame.Position
    local lastMoveOrder = 0

    while autoFarmGrassLineActive do
        if getGrassLineSelectedWorldNumber() ~= targetWorldNumber then
            return "selection_changed"
        end

        local character = Player.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")

        if not character or not humanoid or humanoid.Health <= 0 or not root then
            task.wait(0.2)
            continue
        end

        setGrassLineSpeedForHumanoid(humanoid)

        local distance = (root.Position - targetPosition).Magnitude
        if distance <= 7 then
            humanoid:MoveTo(root.Position)
            return true
        end

        if tick() - lastMoveOrder >= 0.25 then
            lastMoveOrder = tick()
            humanoid:MoveTo(targetPosition)
        end

        task.wait(0.05)
    end

    return false
end

    if not selectedWorld then
        return
    end

    local unlockedWorlds = getGrassLineUnlockedWorlds()
    if selectedWorld > unlockedWorlds then
        return

    if getGrassLineCurrentWorld() ~= selectedWorld then
        task.spawn(function()
            teleportGrassLineToWorld(selectedWorld)
        end)
    end

createOption("Auto Farm Grassline", false, "Farm", function(state)
    autoFarmGrassLineActive = state

    if not state then
        disconnectGrassLineSpeedConnections()

        for humanoid, _ in pairs(grassLineOriginalSpeeds) do
            restoreGrassLineHumanoidSpeed(humanoid)
        end

        local character = Player.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if humanoid and grassLineOriginalSpeeds[humanoid] == nil then
            pcall(function()
                humanoid.WalkSpeed = 28
            end)
        end

        return
    end

    setupGrassLineCharacterSpeed()

    task.spawn(function()
        while autoFarmGrassLineActive do
            local selectedWorld = getGrassLineSelectedWorldNumber()
            local unlockedWorlds = getGrassLineUnlockedWorlds()

            if not selectedWorld then
                task.wait(0.25)
                continue
            end

            if selectedWorld > unlockedWorlds then
                task.wait(0.25)
                continue
            end

            local currentWorld = getGrassLineCurrentWorld()
            if currentWorld ~= selectedWorld then
                teleportGrassLineToWorld(selectedWorld)
                task.wait(0.25)
                continue
            end

            local character = Player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                setGrassLineSpeedForHumanoid(humanoid)
            end

            executeGrassLineSpawnScript()
            task.wait(0.75)

            local targetCoordinate = GRASSLINE_COORDINATES[selectedWorld] or GRASSLINE_COORDINATES[1]
            local walkResult = walkGrassLineToCoordinate(targetCoordinate, selectedWorld)

            if walkResult == false then
                break
            end

            if walkResult == "selection_changed" then
                task.wait(0.05)
                continue
            end

            executeGrassLineSpawnScript()
            task.wait(0.35)
        end
    end)
end)

local function getLootBasePart(item)
    if not item or not item.Parent then
        return nil
    end

    if item:IsA("BasePart") then
        return item
    end

    if item:IsA("Model") then
        local base = item:FindFirstChild("Base")
        if base and base:IsA("BasePart") then
            return base
        end

        local primary = item.PrimaryPart
        if primary and primary:IsA("BasePart") then
            return primary
        end
    end

    return nil

local function getLootRarity(item)
    if not item then
        return nil
    end

    local lootGui = item:FindFirstChild("LootGui")
    if lootGui and lootGui:IsA("BillboardGui") then
        local rarity = lootGui:FindFirstChild("Rarity")
        if rarity and rarity:IsA("TextLabel") and rarity.Text ~= "" then
            return string.lower(rarity.Text)
        end
    end

    if item:IsA("Model") then
        local base = item:FindFirstChild("Base")
        if base then
            lootGui = base:FindFirstChild("LootGui")
            if lootGui and lootGui:IsA("BillboardGui") then
                local rarity = lootGui:FindFirstChild("Rarity")
                if rarity and rarity:IsA("TextLabel") and rarity.Text ~= "" then
                    return string.lower(rarity.Text)
                end
            end
        end
    end

    local nameLower = string.lower(item.Name or "")
    for _, rarityName in ipairs(officialRarityOptions) do
        if string.find(nameLower, string.lower(rarityName), 1, true) then
            return string.lower(rarityName)
        end
    end

    return nil
end

local function getLootDisplayName(item)
    if not item then
        return nil
    end

    local name = tostring(item.Name or "")
    name = name:gsub("^%d+_", "")
    name = name:gsub("^Loot_", "")
    return name ~= "" and name or nil
end

local function isTruthValue(value)
    if value == true then
        return true
    end
    if type(value) == "string" then
        local lower = string.lower(value)
        return lower == "true" or lower == "yes" or lower == "1" or lower == "collected" or lower == "owned" or lower == "found" or lower == "unlocked"
    end
    if type(value) == "number" then
        return value ~= 0
    end
    return false
end

-- A versão atual do arquivo não expõe uma API explícita da Index.
-- Portanto a checagem usa os sinais disponíveis no objeto e na PlayerGui.
local function isLootAlreadyInIndex(item)
    if not item then
        return false
    end

    local attributeNames = {
        "InIndex", "Indexed", "Collected", "Discovered", "Owned", "Found", "Unlocked",
        "IsCollected", "IsDiscovered", "IsOwned", "IsFound", "IndexCollected"
    }

    for _, attributeName in ipairs(attributeNames) do
        local ok, value = pcall(function()
            return item:GetAttribute(attributeName)
        end)
        if ok and isTruthValue(value) then
            return true
        end
    end

    local displayName = string.lower(getLootDisplayName(item) or "")
    if displayName == "" then
        return false
    end

    local playerGui = Player:FindFirstChild("PlayerGui")
    if not playerGui then
        return false
    end

    -- Procura uma entrada da Index pelo nome exato do loot.
    -- Entradas que exibem "???" não contam como descobertas.
    for _, descendant in ipairs(playerGui:GetDescendants()) do
        if descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
            local textValue = string.lower(tostring(descendant.Text or ""))
            if textValue == displayName then
                local parentVisible = true
                local ancestor = descendant
                while ancestor and ancestor ~= playerGui do
                    if ancestor:IsA("GuiObject") and not ancestor.Visible then
                        parentVisible = false
                        break
                    end
                    ancestor = ancestor.Parent
                end

                if parentVisible then
                    local parentName = string.lower(tostring(descendant.Parent and descendant.Parent.Name or ""))
                    local grandParentName = string.lower(tostring(descendant.Parent and descendant.Parent.Parent and descendant.Parent.Parent.Name or ""))
                    if parentName:find("index", 1, true) or parentName:find("collection", 1, true) or parentName:find("loot", 1, true) or grandParentName:find("index", 1, true) or grandParentName:find("collection", 1, true) then
                        return true
                    end
                end
            end
        end
    end

    return false
end

local function teleportToWorldRarityFarm(worldNumber)

    end

    local success = false
    pcall(function()
        local Event = ReplicatedStorage.Packages._Index["acecateer_knit@1.7.2"].knit.Services.WorldService.RF.TeleportToWorld
        if Event then
            Event:InvokeServer(worldNumber)
            success = true
        end
    end)

    if success then
        local timeout = 0
        repeat
            task.wait(0.1)
            timeout += 0.1
        until not autoFarmLootRarityActive or getCurrentWorldNumber() == worldNumber or timeout >= 4
    end

    return success
end

local function getBackpackCapacity()

    local current, maximum = backpackText:match("(%d+)/(%d+)")

    maximum = tonumber(maximum) or 5
    return current, maximum
end

local function collectNearbyLootPrompts()
    local Character = Player.Character
    local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HumanoidRootPart then
        return
    end

    local ProximityPromptService = game:GetService("ProximityPromptService")
    for _, prompt in ipairs(workspace:GetDescendants()) do
        if autoFarmLootRarityActive and prompt:IsA("ProximityPrompt") and prompt.Enabled then
            local promptName = string.lower(prompt.Name or "")
            local actionText = string.lower(prompt.ActionText or "")
            local isPickup =
                promptName == "pickupprompt" or
                promptName:find("pickup", 1, true) or
                promptName:find("pick", 1, true) or
                promptName:find("collect", 1, true) or
                promptName:find("grab", 1, true) or
                actionText:find("pickup", 1, true) or
                actionText:find("pick up", 1, true) or
                actionText:find("collect", 1, true) or
                actionText:find("grab", 1, true) or
                prompt:GetAttribute("LootPickupPrompt") == true

            if isPickup then
                local parent = prompt.Parent
                local position
                if parent and parent:IsA("BasePart") then
                    position = parent.Position
                elseif parent and parent:IsA("Attachment") then
                    position = parent.WorldPosition
                end

                if position and (HumanoidRootPart.Position - position).Magnitude <= prompt.MaxActivationDistance then
                    pcall(function() prompt.HoldDuration = 0 end)
                    pcall(function() fireproximityprompt(prompt) end)
                    pcall(function() prompt:Trigger(Player) end)
                    pcall(function() ProximityPromptService:PromptTriggered(prompt, Player) end)
                    pcall(function()
                        prompt:InputHoldBegin()
                        task.wait(0.01)
                        prompt:InputHoldEnd()
                    end)
                end
            end
        end
    end
end

local function findRarityLootTarget(selectedRarities, currentWorld)
    local zonesFolder = workspace:FindFirstChild("Zones")
    if not zonesFolder then
        return nil
    end

    local selectedMap = {}
    for _, rarity in ipairs(selectedRarities) do
        selectedMap[string.lower(tostring(rarity))] = true
    end

    local bestTarget = nil
    local bestDistance = math.huge
    local Character = Player.Character
    local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
    if not HumanoidRootPart then
        return nil
    end

    for _, zone in ipairs(zonesFolder:GetChildren()) do
        if zone:IsA("Folder") or zone:IsA("Model") then
            for _, descendant in ipairs(zone:GetDescendants()) do
                if descendant:IsA("Model") then
                    local base = getLootBasePart(descendant)
                    if base then
                        local rarity = getLootRarity(descendant)
                        local rarityWorld = getRarityWorld(rarity)
                        if rarity and selectedMap[rarity] and rarityWorld == currentWorld and not isLootAlreadyInIndex(descendant) then
                            local distance = (HumanoidRootPart.Position - base.Position).Magnitude
                            if distance < bestDistance then
                                bestDistance = distance
                                bestTarget = descendant
                            end
                        end
                    end
                end
            end
        end
    end

    return bestTarget
end

createOption("Auto Farm Loot Rarity", false, "Farm", function(state)
    autoFarmLootRarityActive = state

    if not state then
        return
    end

    task.spawn(function()
        local currentTarget = nil
        local lastTargetPosition = nil
        local zonePositioned = false
        local spawnCooldown = false

        while autoFarmLootRarityActive do
            local selectedRarities = optionStates["RarityWorldSelector1"]
            if type(selectedRarities) ~= "table" or #selectedRarities == 0 then
                currentTarget = nil
                task.wait(0.25)
                continue
            end

            local Character = Player.Character
            local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
            if not HumanoidRootPart then
                task.wait(0.5)
                continue
            end

            local highestUnlockedWorld = getUnlockedWorldCount()
            local currentWorld = getCurrentWorldNumber()

            -- Monta os mundos das raridades selecionadas e mantém apenas os mundos desbloqueados.
            local targetWorlds = {}
            local seenWorlds = {}
            for _, rarityName in ipairs(selectedRarities) do
                local rarityWorld = getRarityWorld(rarityName)
                if rarityWorld and rarityWorld <= highestUnlockedWorld and not seenWorlds[rarityWorld] then
                    seenWorlds[rarityWorld] = true
                    table.insert(targetWorlds, rarityWorld)
                end
            end

            if #targetWorlds == 0 then
                task.wait(0.5)
                continue
            end

            local currentWorldIsTarget = seenWorlds[currentWorld] == true
            if not currentWorldIsTarget then
                currentTarget = nil
                zonePositioned = false
                teleportToWorldRarityFarm(targetWorlds[1])
                task.wait(0.75)
                continue
            end

            local currentCap, maxCap = getBackpackCapacity()
            if currentCap >= maxCap then
                currentTarget = nil
                zonePositioned = false
                if not spawnCooldown then
                    spawnCooldown = true
                    teleportToSpawnForRarityFarm()
                end

                repeat
                    task.wait(0.35)
                    currentCap = select(1, getBackpackCapacity())
                until not autoFarmLootRarityActive or currentCap < maxCap

                spawnCooldown = false
                continue
            end

            -- Quando muda de mundo, posiciona na área inicial do mesmo padrão do Auto Best Loots.
            if not zonePositioned then
                local initialZoneCFrames = {
                    [1] = CFrame.new(686.817932, 34.6999893, 0.555465698),
                    [2] = CFrame.new(934.442749, 32.9000015, 1706.30005),
                    [3] = CFrame.new(932.41803, 34.6999893, 3460.30542),
                    [4] = CFrame.new(932.41803, 34.6999893, 5210.30566)
                }
                local destination = initialZoneCFrames[targetWorld] or initialZoneCFrames[1]
                HumanoidRootPart.CFrame = destination
                zonePositioned = true
                task.wait(0.35)
            end

            -- Mantém o jogador no loot atual até ele desaparecer ou entrar na Index.
            if currentTarget and currentTarget.Parent and not isLootAlreadyInIndex(currentTarget) then
                local base = getLootBasePart(currentTarget)
                local targetRarity = getLootRarity(currentTarget)
                if base and targetRarity and getRarityWorld(targetRarity) == currentWorld then
                    local pos = base.Position + Vector3.new(0, 3, 0)
                    if not lastTargetPosition or (lastTargetPosition - base.Position).Magnitude > 1 then
                        HumanoidRootPart.CFrame = CFrame.new(pos)
                        lastTargetPosition = base.Position
                    else
                        HumanoidRootPart.CFrame = CFrame.new(pos)
                    end
                    collectNearbyLootPrompts()
                    task.wait(0.12)
                    continue
                end
            end

            currentTarget = findRarityLootTarget(selectedRarities, currentWorld)
            lastTargetPosition = nil

            if currentTarget then
                local base = getLootBasePart(currentTarget)
                if base then
                    HumanoidRootPart.CFrame = CFrame.new(base.Position + Vector3.new(0, 3, 0))
                    lastTargetPosition = base.Position
                    collectNearbyLootPrompts()
                end
            else
                -- Sem loot disponível neste mundo: passa para o próximo mundo selecionado.
                local nextWorld = nil
                for index, worldNumber in ipairs(targetWorlds) do
                    if worldNumber == currentWorld then
                        nextWorld = targetWorlds[index + 1] or targetWorlds[1]
                        break
                    end
                end

                if nextWorld and nextWorld ~= currentWorld then
                    currentTarget = nil
                    zonePositioned = false
                    teleportToWorldRarityFarm(nextWorld)
                    task.wait(0.75)
                else
                    task.wait(0.35)
                end
            end
        end
    end)
end)

local function hexToColor3(value, fallback)
    if type(value) ~= "string" then
        return fallback
    end

    local hex = value:gsub("#", ""):gsub("%s+", "")
    if #hex == 3 then
        hex = hex:sub(1, 1) .. hex:sub(1, 1)
            .. hex:sub(2, 2) .. hex:sub(2, 2)
            .. hex:sub(3, 3) .. hex:sub(3, 3)
    end

    if #hex ~= 6 or not hex:match("^[%x]+$") then
        return fallback
    end

    local r = tonumber(hex:sub(1, 2), 16)
    local g = tonumber(hex:sub(3, 4), 16)
    local b = tonumber(hex:sub(5, 6), 16)
    if not r or not g or not b then
        return fallback
    end

    return Color3.fromRGB(r, g, b)
end

local function color3ToHex(color)
    return string.format("#%02X%02X%02X",
        math.floor(color.R * 255 + 0.5),
        math.floor(color.G * 255 + 0.5),
        math.floor(color.B * 255 + 0.5)
    )
end

    for _, value in ipairs(espSelection) do
        if value == optionName then
            return true
        end
    end
    return false
end

local function isEnemyPlayer(targetPlayer)
    if targetPlayer == Player then
        return false
    end

    if Player.Team ~= nil then
        return targetPlayer.Team ~= Player.Team
    end

    if targetPlayer.Team ~= nil then
        return true
    end

    return targetPlayer.Neutral ~= Player.Neutral
end

local function destroyEspEntry(targetPlayer)
    local entry = espEntries[targetPlayer]
    if not entry then
        return
    end

    if entry.highlight then
        pcall(function() entry.highlight:Destroy() end)
    end
    if entry.tracer then
        pcall(function() entry.tracer:Destroy() end)
    end
    if entry.container then
        pcall(function() entry.container:Destroy() end)
    end
    espEntries[targetPlayer] = nil
end

local function updateEspEntry(targetPlayer)
    local character = targetPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local entry = espEntries[targetPlayer]

    if not entry then
        createEspEntry(targetPlayer)
        entry = espEntries[targetPlayer]
    end

    if not espEnabled or not character or not humanoid or humanoid.Health <= 0 or not root then
        entry.container.Visible = false
        entry.highlight.Enabled = false
        entry.highlight.Parent = nil
        entry.tracer.Visible = false
        return
    end

    local enemyOnly = isEspSelected("Enemy ESP")
    local enemyTarget = isEnemyPlayer(targetPlayer)
    local passesFilter = not enemyOnly or enemyTarget

    if isEspSelected("Tracer ESP") and enemyTarget then
        local camera = workspace.CurrentCamera
        local localCharacter = Player.Character
        local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
        if camera and localRoot then
            local startPoint, startOnScreen = camera:WorldToViewportPoint(localRoot.Position)
            local endPoint, endOnScreen = camera:WorldToViewportPoint(root.Position)
            if startPoint.Z > 0 and endPoint.Z > 0 and (startOnScreen or endOnScreen) then
                local startVec = Vector2.new(startPoint.X, startPoint.Y)
                local endVec = Vector2.new(endPoint.X, endPoint.Y)
                local delta = endVec - startVec
                local length = delta.Magnitude
                if length > 1 then
                    entry.tracer.BackgroundColor3 = espEnemyColor
                    entry.tracer.Position = UDim2.fromOffset(
                        math.floor((startVec.X + endVec.X) * 0.5 + 0.5),
                        math.floor((startVec.Y + endVec.Y) * 0.5 + 0.5)
                    )
                    entry.tracer.Size = UDim2.fromOffset(math.floor(length + 0.5), math.max(1, rpx(2)))
                    entry.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                    entry.tracer.Visible = true
                else
                    entry.tracer.Visible = false
                end
            else
                entry.tracer.Visible = false
            end
        else
            entry.tracer.Visible = false
        end
    else
        entry.tracer.Visible = false
    end

    if isEspSelected("Player ESP") then
        local localCharacter = Player.Character
        local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
        if localRoot then
            local auraColor = isEnemyPlayer(targetPlayer) and espEnemyColor or espAllyColor
            entry.highlight.FillColor = auraColor
            entry.highlight.OutlineColor = auraColor
            entry.highlight.Parent = character
            entry.highlight.Enabled = true
        else
            entry.highlight.Enabled = false
            entry.highlight.Parent = nil
        end
    else
        entry.highlight.Enabled = false
        entry.highlight.Parent = nil
    end

    local showName = passesFilter and isEspSelected("Name ESP")
    local showHealth = passesFilter and isEspSelected("Health ESP")
    local showDistance = passesFilter and isEspSelected("Distance ESP")
    local showBox = passesFilter and isEspSelected("Box ESP")

    local showOverlay = showName or showHealth or showDistance or showBox
    entry.container.Visible = showOverlay
    if not showOverlay then
        return
    end

    local camera = workspace.CurrentCamera
    if not camera then
        entry.container.Visible = false
        return
    end

    local ok, boundingCFrame, boundingSize = pcall(function()
        return character:GetBoundingBox()
    end)
    if not ok then
        entry.container.Visible = false
        return
    end

    local half = boundingSize * 0.5
    local corners = {
        boundingCFrame * CFrame.new(-half.X, -half.Y, -half.Z),
        boundingCFrame * CFrame.new(-half.X, -half.Y, half.Z),
        boundingCFrame * CFrame.new(-half.X, half.Y, -half.Z),
        boundingCFrame * CFrame.new(-half.X, half.Y, half.Z),
        boundingCFrame * CFrame.new(half.X, -half.Y, -half.Z),
        boundingCFrame * CFrame.new(half.X, -half.Y, half.Z),
        boundingCFrame * CFrame.new(half.X, half.Y, -half.Z),
        boundingCFrame * CFrame.new(half.X, half.Y, half.Z)
    }

    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local anyVisible = false

    for _, corner in ipairs(corners) do
        local viewportPoint, onScreen = camera:WorldToViewportPoint(corner.Position)
        if viewportPoint.Z > 0 then
            minX = math.min(minX, viewportPoint.X)
            minY = math.min(minY, viewportPoint.Y)
            maxX = math.max(maxX, viewportPoint.X)
            maxY = math.max(maxY, viewportPoint.Y)
            anyVisible = anyVisible or onScreen
        end
    end

    if not anyVisible or minX == math.huge then
        entry.container.Visible = false
        return
    end

    local width = math.max(2, maxX - minX)
    local height = math.max(2, maxY - minY)
    entry.container.Position = UDim2.fromOffset(math.floor(minX + 0.5), math.floor(minY + 0.5))
    entry.container.Size = UDim2.fromOffset(math.floor(width + 0.5), math.floor(height + 0.5))

    entry.box.Visible = showBox
    entry.box.Position = UDim2.fromOffset(0, 0)
    entry.box.Size = UDim2.fromScale(1, 1)

    local health = humanoid.MaxHealth > 0 and math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1) or 0
    local healthPercent = health * 100
    local distance = 0
    local localRoot = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if localRoot then
        distance = (localRoot.Position - root.Position).Magnitude
    end

    local infoParts = {}
    if showName then
        table.insert(infoParts, targetPlayer.DisplayName ~= "" and targetPlayer.DisplayName or targetPlayer.Name)
    end
    if showHealth then
        table.insert(infoParts, string.format("%d%%", math.floor(healthPercent + 0.5)))
    end
    if showDistance then
        table.insert(infoParts, string.format("%dM", math.floor(distance + 0.5)))
    end

    local equippedTool = character:FindFirstChildOfClass("Tool")
    local showWeapon = passesFilter and isEspSelected("Weapon ESP") and equippedTool ~= nil
    if showWeapon then
        table.insert(infoParts, equippedTool.Name)
    end

    entry.text.Visible = #infoParts > 0
    entry.text.Text = table.concat(infoParts, " | ")
    entry.text.TextColor3 = espTextColor

    entry.healthBackground.Visible = showHealth and showBox
    if showHealth and showBox then
        entry.healthBackground.Size = UDim2.fromOffset(math.max(4, rpx(5)), math.max(15, math.floor(height)))
        entry.healthBackground.Position = UDim2.fromOffset(-rpx(8), 0)
        entry.healthFill.Size = UDim2.new(1, 0, health, 0)
        if healthPercent > 60 then
            entry.healthFill.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
        elseif healthPercent > 30 then
            entry.healthFill.BackgroundColor3 = Color3.fromRGB(255, 165, 0)
        else
            entry.healthFill.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
        end
    end
end

updateSearchFilter = function()
    local searchText = SearchTextBox.Text:lower()
    local isSearching = searchText ~= ""

    for _, element in pairs(MainContent:GetChildren()) do
        if element:IsA("GuiObject") and element.Name ~= "UIListLayout" then
            local tabName = optionTabMap[element]
            local elementName = optionElementsMap[element] or ""
            local headerKey = categoryHeaderKeys[element]
            local categoryKey = elementCategoryMap[element]

            if isSearching then
                if headerKey then
                    element.Visible = false
                elseif searchOnlyElements[element] then
                    element.Visible = elementName:lower():find(searchText, 1, true) ~= nil
                else
                    element.Visible = elementName ~= "" and elementName:lower():find(searchText, 1, true) ~= nil
                end
            elseif tabName == activeTab then
                if headerKey then
                    element.Visible = true
                elseif searchOnlyElements[element] then
                    element.Visible = false
                elseif categoryKey then
                    if elementName == "" then
                        element.Visible = categoryStates[categoryKey] == true and (element.Size.Y.Offset > 0 or element.Size.Y.Scale > 0)
                    else
                        element.Visible = categoryStates[categoryKey] == true
                    end
                else
                    element.Visible = true
                end
            else
                element.Visible = false
            end
        end
    end

    refreshMainCanvas()
end

SearchTextBox:GetPropertyChangedSignal("Text"):Connect(function()
    updateSearchFilter()
end)

local function updateFloatingButtonPosition(input)
    if activeDragInput and dragStartBtn and startPosBtn then
        local delta = input.Position - dragStartBtn

        if delta.Magnitude > 8 then
            draggingBtn = true
            buttonMoved = true
        end

        if draggingBtn then
            FloatingButton.Position = UDim2.new(startPosBtn.X.Scale, startPosBtn.X.Offset + delta.X, startPosBtn.Y.Scale, startPosBtn.Y.Offset + delta.Y)
        end
    end
end

FloatingButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingBtn = false
        buttonMoved = false
        activeDragInput = input
        dragStartBtn = input.Position
        startPosBtn = FloatingButton.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if activeDragInput then
        if input == activeDragInput or input.UserInputType == Enum.UserInputType.MouseMovement then
            updateFloatingButtonPosition(input)
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if activeDragInput and (input == activeDragInput or input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        if not buttonMoved then
            toggleMenu()
        end

        draggingBtn = false
        buttonMoved = false
        activeDragInput = nil
        dragStartBtn = nil
        startPosBtn = nil
    end
end)
